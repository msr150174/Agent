## 六、常用检索配方

```sql
-- ① 学校名片
SELECT * FROM v_school_card;

-- ② 某专业某学期学什么（专业代码 510203=软件技术 510201=计算机应用 510208=虚拟现实）
SELECT semester, module, course_name, credit, hours, schedule, assessment
FROM v_major_timeline WHERE major_code='510203' AND semester=3;

-- ③ 某专业所有考试课（复习/挂科风险）
SELECT * FROM v_major_exam_course WHERE major_code='510201';

-- ④ 某岗位的完整卡片（职责/技能/证书/对口专业）
SELECT * FROM v_position_card WHERE name LIKE '%测试%';

-- ⑤ 薪资速查（必须带口径与期间，禁止改写成「月薪 X 起」）
SELECT * FROM v_salary_lookup WHERE job_name='质检员';

-- ⑥ 某专业能不能专升本/考编
SELECT path, coverage, note FROM maj_path_coverage WHERE major_id='SE';

-- ⑦ 答「未覆盖」之前先查缺口表
SELECT gap_title, status, advise FROM v_gap_all WHERE gap_title LIKE '%薪资%';

-- ⑧ 全文召回（seg 需逐字加空格并加引号）
SELECT s.doc_id, s.title, snippet(kb_search,3,'…','…',24)
FROM kb_search f JOIN kb_section s ON s.sec_id=f.sec_id
WHERE f.seg MATCH '" 宿 舍 矛 盾 "' ORDER BY f.rank LIMIT 5;
```
